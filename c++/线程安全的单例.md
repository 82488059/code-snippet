#ifndef __SINGLETONSAFET_H__  
#define __SINGLETONSAFET_H__  
#include <Windows.h>  
#include <iostream>  

/////////////////////////////////////////////////////////////////////
// 工程: 
// 作者: zt
// 描述: 临界区
// 主要函数：
// 日期: 2015.08.31
// 版本: 1.0

/////////////////////////////////////////////////////////////////////
class CLocker
{
public:
    CLocker()
    {
        InitializeCriticalSection(&m_cs);
    }
    ~CLocker()
    {
        DeleteCriticalSection(&m_cs);
    }
    void Lock()
    {
        EnterCriticalSection(&m_cs);
    }
    void UnLock()
    {
        LeaveCriticalSection(&m_cs);
    }
private:
    CRITICAL_SECTION m_cs;

};
/////////////////////////////////////////////////////////////////////
// 工程: 
// 作者: zt
// 描述: 单例模版类
// 主要函数：
// 日期: 2015.08.31
// 版本: 1.0
// 修改:
/////////////////////////////////////////////////////////////////////
template <class T>
class SingletonSafeT
{
public:
    static T* GetInstance()
    {
        if (__instance == 0)
        {
            Lock.Lock();
            if (__instance == 0)
            {
                __instance = new T();
            }
            Lock.UnLock();
        }
        return __instance;
    }

    SingletonSafeT(){};
    ~SingletonSafeT(){};
private:
    static T* __instance;
    static CLocker Lock;
};
template<class T>
T* SingletonSafeT<T>::__instance = 0;
template<class T>
CLocker SingletonSafeT<T>::Lock;


#endif  // __SINGLETONSAFET_H__
