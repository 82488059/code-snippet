```c++
#pragma once
/////////////////////////////////////////////////////////////////////
// 工程: 
// 作者: zt
// 描述: 临界区自动加解锁
// 主要函数：
// 日期: 2014.12.25
// 版本: 1.0
// 修改:
/////////////////////////////////////////////////////////////////////
class CAutoLock
{
public:

    CAutoLock(CRITICAL_SECTION* pcs):m_pcs(pcs)
    {
        EnterCriticalSection(m_pcs);
    }
    ~CAutoLock()
    {
        LeaveCriticalSection(m_pcs);
    }
private:
    
    CRITICAL_SECTION* m_pcs;
};
```
